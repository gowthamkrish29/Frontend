async function fetchUserData(userId) {
  try {
    const res = await fetch(`https://api.github.com/users/${userId}`);
    if (!res.ok) throw new Error("Not Found");
    return await res.json();
  } catch (err) {
    console.error("Fetch error:", err.message);
  }
}

function twoSum(nums, target) { const map = new Map(); for (let i = 0; i < nums.length; i++) { const diff = target - nums[i]; if (map.has(diff)) return [map.get(diff), i]; map.set(nums[i], i); } return []; } console.log(twoSum([2, 7, 11, 15], 9));
Execution Output: [0, 1] (Target 9 found in 0.4ms)
